# Data Contracts
### AI Discovery Engine — Myntra
### Backend ↔ Frontend JSON Specification

---

> **Purpose:** This document defines the exact JSON shape for every file the pipeline exports and the dashboard consumes. Both pipeline (Python) and dashboard (Next.js) developers must treat these schemas as the contract — any change here must be reflected on both sides.

---

## 1. File Map

| File | Generated By | Consumed By | Description |
|---|---|---|---|
| `summary.json` | `quantification/export.py` | Dashboard home page | Executive summary stats |
| `q1.json` – `q10.json` | `quantification/export.py` | Dashboard question pages | Per-question quantified breakdowns |
| `systemic_gaps.json` | `quantification/export.py` | Dashboard systemic gaps page | Secondary source analysis |
| `corpus_meta.json` | `cleaning/pipeline.py` | Dashboard footer / about | Corpus-level metadata |

All files live in: `dashboard/public/data/`

## 6. TypeScript Interfaces (for Dashboard)

For type safety in the Next.js dashboard:

```typescript
// types/data.ts

interface KPICard {
  id: string;
  label: string;
  value: number;
  unit: 'count' | 'percent';
  sparkline: number[];
  trend: 'up' | 'down' | 'stable';
  trend_pct: number;
}

interface SourceItem {
  source: string;
  count: number;
  pct: number;
  color: string;
}

interface OpportunityItem {
  rank: number;
  label: string;
  question_id: number;
  pct: number;
  count: number;
  avg_confidence: number;
  impact_score: number;
}

interface SummaryData {
  generated_at: string;
  pipeline_version: string;
  kpi_cards: KPICard[];
  source_distribution: SourceItem[];
  top_opportunities: OpportunityItem[];
  overall_confidence: number;
  primary_signal_docs: number;
  secondary_signal_docs: number;
  no_signal_docs: number;
}

interface BreakdownItem {
  label: string;
  tag: string;
  count: number;
  pct: number;
  avg_confidence: number;
  color: string;
}

interface Quote {
  text: string;
  source: string;
  source_id: string;
  date: string;
  confidence: number;
  tags: string[];
  segment: string;
}

interface TemporalPoint {
  month: string;
  count: number;
  pct: number;
}

interface QuestionData {
  question_id: number;
  question_text: string;
  question_short: string;
  total_relevant_docs: number;
  avg_confidence: number;
  breakdown: BreakdownItem[];
  segment_splits: Record<string, BreakdownItem[]>;
  source_attribution: SourceAttributionItem[];
  key_quotes: Quote[];
  temporal_trend: TemporalPoint[];
}
```
